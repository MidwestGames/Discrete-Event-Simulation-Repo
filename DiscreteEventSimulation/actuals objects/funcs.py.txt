import numpy as np
import pandas as pd

def CalculatePadding(header : str, data):
    pad = -1

    if(len(header) > len(str(data))):
       pad = len(header)
    else:
        pad = len(str(data))


    return pad

def GetPaddedString(value, pad : int):
    result = ""

    p = pad - len(str(value))

    result += str(value)

    for s in range(1, p+1):
        result += " "

    result += "|"
    return result
